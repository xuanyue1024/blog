```html-demo
<main>
    <header>
      <h2>P2P 全连接拓扑 (Full Mesh)</h2>
      <p>这是一个完全使用<strong>语义化 HTML</strong>编写的演示。通过切换 <code>&lt;head&gt;</code> 中的 CSS 链接，你可以看到它在不同无类框架下的表现。</p>
    </header>

    <section>
      <form onsubmit="event.preventDefault();">
        <fieldset>
          <label for="nodeCount">
            当前节点数: <mark id="nodeDisplay">5</mark> | 
            需建立的连接数: <mark id="connDisplay">10</mark>
          </label>
          <input type="range" id="nodeCount" min="2" max="15" value="5" style="width: 100%;">
        </fieldset>
      </form>
    </section>

    <figure>
      <canvas id="topologyCanvas" width="400" height="400"></canvas>
      <figcaption>全网状拓扑结构的复杂度呈二次方增长：N(N-1)/2</figcaption>
    </figure>
  </main>

  <script>
    const canvas = document.getElementById('topologyCanvas');
    const ctx = canvas.getContext('2d');
    const countInput = document.getElementById('nodeCount');
    const nodeDisplay = document.getElementById('nodeDisplay');
    const connDisplay = document.getElementById('connDisplay');

    function drawNetwork(nodeCount) {
      const width = canvas.width;
      const height = canvas.height;
      const radius = 140;
      const centerX = width / 2;
      const centerY = height / 2;

      ctx.clearRect(0, 0, width, height);

      const nodes = [];
      for(let i = 0; i < nodeCount; i++) {
        const angle = (i / nodeCount) * 2 * Math.PI - Math.PI / 2;
        nodes.push({ x: centerX + radius * Math.cos(angle), y: centerY + radius * Math.sin(angle) });
      }

      // 【黑科技】获取当前环境的文字颜色，让线条自适应各种框架的暗黑/明亮模式
      const textColor = getComputedStyle(document.body).color || '#666';

      // 1. 画连线
      ctx.beginPath();
      ctx.strokeStyle = textColor;
      ctx.globalAlpha = 0.2; // 降低透明度避免线条干扰
      ctx.lineWidth = 1.5;
      for(let i = 0; i < nodeCount; i++) {
        for(let j = i + 1; j < nodeCount; j++) {
          ctx.moveTo(nodes[i].x, nodes[i].y);
          ctx.lineTo(nodes[j].x, nodes[j].y);
        }
      }
      ctx.stroke();

      // 2. 画节点
      ctx.globalAlpha = 1.0;
      ctx.fillStyle = '#007bff'; // 节点使用通用科技蓝
      for(let i = 0; i < nodeCount; i++) {
        ctx.beginPath();
        ctx.arc(nodes[i].x, nodes[i].y, 8, 0, 2 * Math.PI);
        ctx.fill();
      }

      nodeDisplay.textContent = nodeCount;
      connDisplay.textContent = (nodeCount * (nodeCount - 1)) / 2;
    }

    countInput.addEventListener('input', (e) => drawNetwork(parseInt(e.target.value)));
    
    // 稍微延迟渲染，确保外部 CSS 加载完毕并应用，以便 Canvas 获取到正确的环境文字颜色
    setTimeout(() => drawNetwork(parseInt(countInput.value)), 100);
  </script>
```
